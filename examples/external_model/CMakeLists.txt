cmake_minimum_required(VERSION 4.4.3)
project(Elf3DExternalModel LANGUAGES C CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/lib/$<CONFIG>")
set(ELF3D_SOURCE_DIR "" CACHE PATH "Path to the shared Elf3D source checkout")
if(NOT EXISTS "${ELF3D_SOURCE_DIR}/include/elf3d/model.h")
    message(FATAL_ERROR "Set ELF3D_SOURCE_DIR to the Elf3D source checkout")
endif()
include(CTest)
set(ELF3D_BUILD_ENGINE OFF CACHE BOOL "" FORCE)
set(ELF3D_BUILD_APP OFF CACHE BOOL "" FORCE)
set(ELF3D_BUILD_VIEWER OFF CACHE BOOL "" FORCE)
set(ELF3D_BUILD_CMD OFF CACHE BOOL "" FORCE)
set(ELF3D_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${ELF3D_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/elf3d" EXCLUDE_FROM_ALL)

# Copied reference code becomes ordinary application-owned source.
configure_file("${ELF3D_SOURCE_DIR}/apps/cmd/src/rotate_document.h" copied/rotate_document.h COPYONLY)
configure_file("${ELF3D_SOURCE_DIR}/apps/cmd/src/rotate_document.cpp" copied/rotate_document.cpp COPYONLY)
add_executable(elf3d_external_model main.cpp "${PROJECT_BINARY_DIR}/copied/rotate_document.cpp")
target_include_directories(elf3d_external_model PRIVATE "${PROJECT_BINARY_DIR}/copied")
target_link_libraries(elf3d_external_model PRIVATE elf3d::model)
set_target_properties(elf3d_external_model PROPERTIES CXX_EXTENSIONS OFF)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT elf3d_external_model)
if(MSVC)
    target_compile_options(elf3d_external_model PRIVATE /W4 /WX /permissive-)
endif()
if(BUILD_TESTING)
    add_test(NAME external_model_processing COMMAND elf3d_external_model
        "${ELF3D_SOURCE_DIR}/tests/fixtures/elf3d_smoke/elf3d_smoke.gltf"
        "${PROJECT_BINARY_DIR}/bin/$<CONFIG>/processed.glb")
endif()
